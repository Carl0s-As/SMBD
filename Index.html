Guía de código SQL: DDL y DML

Sintaxis base estilo MySQL; las variantes por SMBD están en la sección 5. Cada bloque indica con qué se combina. Los nombres (t, col, padre, hija…) son genéricos.

1. DDL: definir y cambiar la estructura

CREATE TABLE

Se combina con: tipos de dato, PRIMARY KEY, FOREIGN KEY (la tabla padre debe existir antes), NOT NULL, UNIQUE, CHECK, DEFAULT.

CREATE TABLE padre (
  id       VARCHAR(5)   PRIMARY KEY,
  nombre   VARCHAR(30)  NOT NULL UNIQUE,
  vigente  TINYINT      DEFAULT 1 CHECK (vigente IN (0,1))
);
CREATE TABLE hija (
  cod      VARCHAR(10),
  nombre   VARCHAR(30)  NOT NULL,
  precio   DECIMAL(8,2) DEFAULT 0,
  id_padre VARCHAR(5),
  PRIMARY KEY (cod),
  CONSTRAINT fk_hija_padre FOREIGN KEY (id_padre) REFERENCES padre(id)
);
-- PK compuesta (tabla de detalle, relación M:N)
CREATE TABLE detalle (
  folio INT, cod VARCHAR(10), cant INT NOT NULL,
  PRIMARY KEY (folio, cod),
  FOREIGN KEY (folio) REFERENCES cabecera(folio),
  FOREIGN KEY (cod)   REFERENCES hija(cod)
);

Tipos: INT DECIMAL(p,s) VARCHAR(n) CHAR(n) DATE TIME DATETIME BOOLEAN. NOT NULL solo a nivel de columna; PK/UNIQUE/FK compuestas a nivel de tabla.

ALTER TABLE

Se combina con: ADD / DROP / MODIFY, ADD CONSTRAINT. Después suele venir un UPDATE (poblar la columna nueva) o un INSERT.

ALTER TABLE t ADD c1 VARCHAR(10) NULL, ADD c2 TIME;       -- columnas nuevas
ALTER TABLE t DROP COLUMN c1;                              -- quitar columna
ALTER TABLE t MODIFY c1 VARCHAR(30) NOT NULL;              -- cambiar tipo
ALTER TABLE t ADD CONSTRAINT pk_t PRIMARY KEY (id);
ALTER TABLE t ADD CONSTRAINT uq_t UNIQUE (c1);
ALTER TABLE t ADD CONSTRAINT ck_t CHECK (precio >= 0);
ALTER TABLE t ALTER COLUMN c2 SET DEFAULT 0;
-- Relación 1:N sobre tablas ya creadas (la FK va en el lado "muchos")
ALTER TABLE hija ADD id_padre VARCHAR(5);
ALTER TABLE hija ADD CONSTRAINT fk_h FOREIGN KEY (id_padre) REFERENCES padre(id);

CREATE VIEW

Se combina con: cualquier SELECT (JOIN, GROUP BY, HAVING, ORDER BY). Las columnas calculadas necesitan alias. Luego se consulta como una tabla y puede servir de base a otra vista.

CREATE VIEW v_resumen AS
SELECT a.k, COUNT(b.k) AS n, COALESCE(SUM(b.monto),0) AS total
FROM a LEFT JOIN b ON a.k = b.k
GROUP BY a.k;

SELECT v.*, total * 0.10 AS comision FROM v_resumen v;   -- consultar la vista

RENAME · DROP · TRUNCATE · INDEX · COMMENT

RENAME TABLE viejo TO nuevo;              -- MySQL (ver variantes)
DROP VIEW v_resumen;   DROP TABLE hija;   -- hijas primero
DROP DATABASE ejemplo;
TRUNCATE TABLE t;                         -- vacía, conserva estructura, sin WHERE
CREATE UNIQUE INDEX ix_t_c ON t (c);      -- o CREATE INDEX
CREATE ROLE consultas;   CREATE USER susana IDENTIFIED BY 'clave';
COMMENT ON COLUMN t.c IS 'texto';         -- Oracle/PostgreSQL

2. DML: INSERT, UPDATE, DELETE

INSERT

Depende de: tabla y columnas ya creadas, padres insertados antes que hijas (FK). Se combina con: lista de columnas, VALUES o SELECT.

INSERT INTO t (c1, c2) VALUES ('a', 1);                      -- un registro
INSERT INTO t (c1, c2) VALUES ('a', 1), ('b', 2), ('c', 3);  -- varios
INSERT INTO t VALUES ('a', 1, 'x');                          -- todas las columnas, en orden
INSERT INTO t2 (c1, c2) SELECT c1, c2 FROM t1 WHERE cond;    -- desde consulta (sin VALUES)

UPDATE

Se combina con: SET (valor, expresión o subconsulta) y WHERE (comparación, IN, NOT IN, EXISTS). Sin WHERE afecta a todas las filas.

UPDATE t SET c1 = 'x', c2 = 5 WHERE clave = 'k';       -- completar datos por clave
UPDATE t SET precio = precio * 1.2;                     -- todas las filas
UPDATE t SET c = (SELECT AVG(c) FROM t) WHERE c BETWEEN 4000 AND 5000;
-- Marcar según existan (o no) registros en otra tabla: dos UPDATE complementarios
UPDATE t SET flag = 0 WHERE k NOT IN (SELECT k FROM det WHERE k IS NOT NULL);
UPDATE t SET flag = 1 WHERE k IN (SELECT k FROM det);
-- Equivalente con EXISTS
UPDATE t SET flag = 1 WHERE EXISTS (SELECT 1 FROM det d WHERE d.k = t.k);

DELETE

Se combina con: WHERE. Orden: hijas antes que padres. Borrar todo sin perder estructura: DELETE o TRUNCATE tabla por tabla.

DELETE FROM t WHERE sueldo > 20000;
DELETE FROM hija;  DELETE FROM padre;      -- vaciar respetando FK

3. SELECT: la sentencia que se combina con todo

SELECT [DISTINCT] col, expr AS alias, funcion(col)     -- 5º  (qué mostrar)
FROM   t1 [alias]                                      -- 1º  (de dónde)
       [INNER|LEFT|RIGHT|FULL] JOIN t2 ON t1.pk = t2.fk -- 2º  (cómo unir)
WHERE  cond [AND|OR|NOT cond]                          -- 3º  filtra FILAS
GROUP BY col                                           -- 4º  agrupa
HAVING agregado cond                                   -- 4º  filtra GRUPOS
ORDER BY col | alias | nº [ASC|DESC];                  -- 6º  ordena

JOIN

FROM a INNER JOIN b ON a.k = b.k        -- solo coincidencias
FROM a LEFT  JOIN b ON a.k = b.k        -- todo A; sin par, B = NULL
FROM a RIGHT JOIN b ON a.k = b.k        -- todo B
FROM a FULL OUTER JOIN b ON a.k = b.k   -- todo de ambos
FROM a NATURAL JOIN b                   -- une por columnas con el mismo nombre
FROM a CROSS JOIN b                     -- producto cartesiano (filas A x filas B)
FROM a, b WHERE a.k = b.k               -- forma antigua de INNER JOIN
-- Solo las filas de A SIN par en B:
SELECT a.* FROM a LEFT JOIN b ON a.k = b.k WHERE b.k IS NULL;

Condiciones (WHERE / HAVING)

col = 5   col <> 5   col > 5   col <= 5
col BETWEEN 10 AND 20        col NOT BETWEEN 10 AND 20
col IN ('a','b')             col NOT IN (SELECT k FROM t2)
col LIKE 'Ma%'  (% varios, _ uno)      col IS NULL   col IS NOT NULL
EXISTS (SELECT 1 FROM t2 WHERE t2.k = t1.k)     cond1 AND (cond2 OR NOT cond3)

Agregación

COUNT, SUM, AVG, MIN, MAX, VARIANCE. Van en SELECT y HAVING, nunca en WHERE. Toda columna del SELECT que no sea agregado debe estar en GROUP BY.

SELECT k, COUNT(*) AS n, SUM(cant) AS q, AVG(precio) AS prom
FROM det GROUP BY k HAVING SUM(cant) > 2 ORDER BY q DESC;

Subconsultas

WHERE precio > (SELECT AVG(precio) FROM t)                -- escalar
WHERE k IN (SELECT k FROM det)                            -- lista
WHERE EXISTS (SELECT 1 FROM det d WHERE d.k = t.k)        -- correlacionada
FROM (SELECT k, SUM(m) AS tot FROM det GROUP BY k) x      -- en FROM (alias obligatorio)
SELECT nom, (SELECT COUNT(*) FROM det d WHERE d.k = t.k) AS n FROM t;  -- en SELECT

Funciones

CONCAT(nom, ' ', ape) AS completo    UPPER LOWER INITCAP LPAD RPAD LTRIM REPLACE SUBSTR
ABS SQRT EXP LN LOG SIN COS TAN      precio * 1.20 AS con_aumento   'Texto: ' AS etiqueta
SYSDATE   MONTHS_BETWEEN(f1,f2)   ADD_MONTHS(f,n)
TO_CHAR(x)   TO_DATE('01/02/1995','DD/MM/YYYY')   TO_NUMBER('12')   CAST(x AS INT)
COALESCE(SUM(m), 0)      -- reemplaza NULL (IFNULL en MySQL, NVL en Oracle)

4. Combinaciones típicas (DDL + DML juntos)

A. Reporte de TODAS las filas de A, tengan o no detalle en B

LEFT JOIN + GROUP BY + COUNT(columna de B) + SUM + COALESCE. Se puede guardar con CREATE VIEW y luego calcular sobre la vista.

SELECT a.k, CONCAT(a.nom,' ',a.ape) AS nombre, COUNT(b.k) AS n, COALESCE(SUM(b.m),0) AS total
FROM a LEFT JOIN b ON a.k = b.k
GROUP BY a.k, a.nom, a.ape;

B. Ranking filtrado, guardado como vista

JOIN + GROUP BY + SUM + HAVING + ORDER BY DESC → CREATE VIEW.

CREATE VIEW v_top AS
SELECT p.cod, p.nombre, p.precio, SUM(d.cant) AS vendidos
FROM p INNER JOIN d ON p.cod = d.cod
GROUP BY p.cod, p.nombre, p.precio
HAVING SUM(d.cant) > 2
ORDER BY vendidos DESC;

C. Columna nueva + poblarla en dos pasos

ALTER ADD → UPDATE (condición) → UPDATE (condición complementaria).

ALTER TABLE t ADD vigente BOOLEAN;
UPDATE t SET vigente = 0 WHERE k NOT IN (SELECT k FROM det WHERE k IS NOT NULL);
UPDATE t SET vigente = 1 WHERE vigente IS NULL;

D. Nueva tabla catálogo y relación 1:N

CREATE TABLE → INSERT (varias filas) → ALTER ADD columna FK en la hija → UPDATE para asignar.

CREATE TABLE catalogo (id VARCHAR(5) PRIMARY KEY, descripcion VARCHAR(30), vigente TINYINT);
INSERT INTO catalogo VALUES ('A1','Uno',1), ('A2','Dos',1);
ALTER TABLE hija ADD id_cat VARCHAR(5);
ALTER TABLE hija ADD CONSTRAINT fk_cat FOREIGN KEY (id_cat) REFERENCES catalogo(id);
UPDATE hija SET id_cat = 'A1' WHERE cod = 'x';

E. Registro nuevo + completar sus datos después

INSERT INTO prov (rfc, empresa) VALUES ('ABC123', 'Empresa');
ALTER TABLE prov ADD calle VARCHAR(20) NULL, ADD cp INT NULL;   -- ver variantes
UPDATE prov SET calle = 'Reforma', cp = 6600 WHERE rfc = 'ABC123';

F. Vaciar toda la base y luego eliminarla

DELETE FROM detalle; DELETE FROM cabecera; DELETE FROM hija; DELETE FROM padre;
-- o desactivando FK (MySQL): SET FOREIGN_KEY_CHECKS = 0; TRUNCATE ...; SET FOREIGN_KEY_CHECKS = 1;
DROP TABLE detalle; DROP TABLE cabecera; DROP TABLE hija; DROP TABLE padre;

6. DCL y TCL

GRANT SELECT, DELETE ON t TO susana;   GRANT consultas TO susana;   GRANT ALL ON t TO susana;
REVOKE SELECT ON t FROM susana;        REVOKE consultas FROM susana;
COMMIT;      -- confirma los cambios de DML
ROLLBACK;    -- deshace desde el último COMMIT

7. Trampas frecuentes

COUNT(*) vs COUNT(col)
Con LEFT JOIN, COUNT(*) cuenta la fila vacía como 1. Cuenta una columna de la tabla derecha.
NOT IN y NULL
Si la subconsulta devuelve NULL, NOT IN no devuelve filas. Usa IS NOT NULL o NOT EXISTS.
WHERE vs HAVING
WHERE filtra antes de agrupar; HAVING filtra los grupos ya calculados.
Ceros a la izquierda
Un CP como INT pierde el 0 inicial (08620 pasa a 8620). Si importa, usa texto.
Vistas
Alias en toda columna calculada. Una vista se consulta y se combina como cualquier tabla.
Orden con FK
INSERT: padres primero. DELETE y DROP: hijas primero.
INSERT con subconsulta
Se escribe INSERT … SELECT, sin la palabra VALUES.
Longitudes
Verifica que VARCHAR(n) alcance para los datos (RFC, calles, colonias).
